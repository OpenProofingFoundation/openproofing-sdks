cmake_minimum_required(VERSION 3.20)
project(kenshiki_portable LANGUAGES CXX)
add_library(kenshiki_portable STATIC src/portrait.cpp src/auraface_input.cpp src/face_alignment.cpp src/frame_selection.cpp)
target_compile_features(kenshiki_portable PUBLIC cxx_std_20)
target_compile_options(kenshiki_portable PRIVATE -fno-fast-math -ffp-contract=off)
target_include_directories(kenshiki_portable PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(kenshiki_portable PUBLIC Threads::Threads)
if(NOT APPLE)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(kenshiki_portable PUBLIC OpenSSL::Crypto)
endif()
# The governed nist/check.py runner generates authoritative vector data and runs
# the adversarial executable under ASan/UBSan on both Apple and Linux.

# ORT remains an explicit, optional external dependency. No downloads or runtime
# libraries enter the portable package or mobile consumers through this target.
option(KENSHIKI_AURAFACE "Build the pinned AuraFace CPU inference adapter" OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "External ONNX Runtime 1.22.0 installation")
if(KENSHIKI_AURAFACE)
    if(NOT IS_DIRECTORY "${ONNXRUNTIME_ROOT}/include" OR NOT IS_DIRECTORY "${ONNXRUNTIME_ROOT}/lib")
        message(FATAL_ERROR "ONNXRUNTIME_ROOT must name an external ONNX Runtime 1.22.0 installation")
    endif()
    if(NOT EXISTS "${ONNXRUNTIME_ROOT}/VERSION_NUMBER")
        message(FATAL_ERROR "ONNX Runtime version record is required")
    endif()
    file(READ "${ONNXRUNTIME_ROOT}/VERSION_NUMBER" KENSHIKI_ORT_VERSION)
    string(STRIP "${KENSHIKI_ORT_VERSION}" KENSHIKI_ORT_VERSION)
    if(NOT KENSHIKI_ORT_VERSION STREQUAL "1.22.0")
        message(FATAL_ERROR "ONNX Runtime 1.22.0 is required")
    endif()
    find_library(KENSHIKI_ORT_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    add_library(kenshiki_auraface STATIC src/auraface.cpp)
    target_compile_features(kenshiki_auraface PUBLIC cxx_std_20)
    target_include_directories(kenshiki_auraface PUBLIC include PRIVATE "${ONNXRUNTIME_ROOT}/include")
    target_compile_options(kenshiki_auraface PRIVATE -fno-fast-math -ffp-contract=off)
    target_link_libraries(kenshiki_auraface PUBLIC kenshiki_portable "${KENSHIKI_ORT_LIBRARY}")
endif()

add_executable(kenshiki_alignment_tests tests/face_alignment_tests.cpp)
target_link_libraries(kenshiki_alignment_tests PRIVATE kenshiki_portable)
target_compile_options(kenshiki_alignment_tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME face_alignment COMMAND kenshiki_alignment_tests)
add_executable(kenshiki_selection_tests tests/frame_selection_tests.cpp)
target_link_libraries(kenshiki_selection_tests PRIVATE kenshiki_portable)
target_compile_options(kenshiki_selection_tests PRIVATE -UNDEBUG)
add_test(NAME frame_selection COMMAND kenshiki_selection_tests)
